public void MoveRight(int steps, int delayMs = 150)
    {

        for (int i = 0; i < steps; i++)
        {
            horizPos++;
            if (i % 2 == 0) //i is even 
            {
                GoombaLeft();

            }
            else //i is odd
            {
                GoombaRight();
            }
            //PrintGoomba();
            DrawImage();
            Thread.Sleep(delayMs);

        }
    }
    public void MoveLeft(int steps, int delayMs = 150)
    {
        for (int i = 0; i < steps; i++)
        {
            if (horizPos > 0) horizPos--; //decrement, rmv spaces 

            if (i % 2 == 0) GoombaLeft();

            else GoombaRight();

            //PrintGoomba();
            DrawImage();
            Thread.Sleep(delayMs);

        }
    }
    public void Stand()
    {
        GoombaFlat();
        //PrintGoomba();
        DrawImage();
    }


    public void Main()
    {
        //DrawImage();
        MoveRight(15);
        MoveLeft(15);
        Stand();
    }
